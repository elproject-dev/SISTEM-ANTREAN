                          ? Math.round((t.doneAt - t.calledAt) / 1000)
                          : null;

                      return (
                        <tr key={t.id} className="transition-colors hover:bg-slate-50">
                          <td className="px-4 py-3 border-b border-slate-200 text-center text-xs font-semibold text-slate-600">

                          </td>
                          <td className="px-4 py-3 border-b border-slate-200 text-center">
                            <span className="font-black text-base text-primary">{t.displayNumber}</span>
                          </td>
                          <td className="px-4 py-3 border-b border-slate-200">
                            <div className="font-bold text-slate-800 text-xs">
                              {t.customerName || 'Umum'}
                            </div>
                            <div className="text-[10px] text-slate-500 font-medium mt-0.5">
                              {SERVICES.find(s => s.code === t.serviceCode)?.name}
                            </div>
                          </td>
                          <td className="px-4 py-3 border-b border-slate-200 text-center text-xs font-medium text-slate-600">
                            {dur !== null ? (dur < 60 ? `${dur}d` : `${Math.floor(dur / 60)}m ${dur % 60}d`) : '-'}
                          </td>
                          <td className="px-4 py-3 border-b border-slate-200 text-center">
                            <CustomStatusBadge variant={sc.variant} label={sc.label} />
                          </td>
                        </tr>
                      )
                    })
                  )}
                </tbody>
              </table>
            </div>
          </div>

        </div>
      </div>
    </div>
  );
}
